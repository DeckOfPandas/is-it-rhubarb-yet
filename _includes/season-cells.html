{% comment %} The shapes for one chart row (one variety). Expects: variety, label, y, x0, cw, rh. Colour comes from the enclosing element's --c custom property. A fully in-season range is a full-height bar. A "just about" range tapers: rising into an in-season range that follows it, falling out of one that precedes it, and a low band when it touches neither or both. Year ends count as adjacent, so a December ramp can lead into a January season. {% endcomment %} {%- assign h = include.rh -%} {%- assign low = h | times: 0.4 | round -%} {%- assign lowtop = include.y | plus: h | minus: low -%} {%- assign bottom = include.y | plus: h -%} {%- for r in include.variety.weeks -%} {%- assign x1 = r[0] | minus: 1 | times: include.cw | plus: include.x0 -%} {%- assign x2 = r[1] | times: include.cw | plus: include.x0 -%} {%- if r[2] == "in" -%} {{ include.label }}: weeks {{ r[0] }} to {{ r[1] }} {%- else -%} {%- assign before = r[0] | minus: 1 -%}{%- if before == 0 -%}{%- assign before = 52 -%}{%- endif -%} {%- assign after = r[1] | plus: 1 -%}{%- if after == 53 -%}{%- assign after = 1 -%}{%- endif -%} {%- assign rising = false -%}{%- assign falling = false -%} {%- for o in include.variety.weeks -%} {%- if o[2] == "in" -%} {%- if o[0] == after -%}{%- assign rising = true -%}{%- endif -%} {%- if o[1] == before -%}{%- assign falling = true -%}{%- endif -%} {%- endif -%} {%- endfor -%} {%- if rising and falling == false -%} {{ include.label }}: coming in, weeks {{ r[0] }} to {{ r[1] }} {%- elsif falling and rising == false -%} {{ include.label }}: going out, weeks {{ r[0] }} to {{ r[1] }} {%- else -%} {{ include.label }}: just about, weeks {{ r[0] }} to {{ r[1] }} {%- endif -%} {%- endif -%} {%- endfor -%}
